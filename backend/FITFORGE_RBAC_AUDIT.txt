class RolePermission(BasePermission):

    message = "You do not have permission to perform this action."

    def has_permission(self, request, view):

        required_permission = getattr(
            view,
            "required_permission",
            None,
        )

        # If the view doesn't require a permission,
        # allow authenticated users.
        if required_permission is None:
            return True

        if not request.user.is_authenticated:
            return False

        try:

            employee = (
                Employee.objects
                .select_related("role")
                .prefetch_related("role__permissions")
                .get(user=request.user)
            )

        except Employee.DoesNotExist:
            return False

        if employee.role is None:
            return False

        return employee.role.permissions.filter(
            code=required_permission,
            is_active=True,
        ).exists()

from rest_framework.exceptions import PermissionDenied

from .constants import ROLE_LEVELS


class RolePermissionMixin:
    """
    Enterprise RBAC Mixin

    Responsibilities:
    - Enforce minimum role hierarchy
    - Enforce permission-based authorization
    """

    minimum_role_level = 0
    required_permission = None

    def get_employee(self):
        return getattr(self.request.user, "employee_profile", None)

    def get_user_role_level(self):
        user = self.request.user

        if user.is_superuser:
            return ROLE_LEVELS["SUPER_ADMIN"]

        employee = self.get_employee()

        if employee and employee.role:
            return ROLE_LEVELS.get(employee.role.name, 0)

        return 0

    def has_required_permission(self):
        if not self.required_permission:
            return True

        employee = self.get_employee()

        if not employee or not employee.role:
            return False

        return employee.role.permissions.filter(
            code=self.required_permission,
            is_active=True,
        ).exists()

    def initial(self, request, *args, **kwargs):
        super().initial(request, *args, **kwargs)

        if self.get_user_role_level() < self.minimum_role_level:
            raise PermissionDenied(
                "You do not have the required role level."
            )

        if not self.has_required_permission():
            raise PermissionDenied(
                "You do not have the required permission."
            )

from rest_framework.permissions import BasePermission


class EmployeePermission(BasePermission):
    """
    Permission class for Employee APIs.
    """

    ALLOWED_ROLES = {
        "SUPER_ADMIN",
        "ORG_OWNER",
        "BRANCH_MANAGER",
    }

    def has_permission(self, request, view):

        if not request.user.is_authenticated:
            return False

        if request.user.is_superuser:
            return True

        if not hasattr(request.user, "employee_profile"):
            return False

        employee = request.user.employee_profile

        if employee.role is None:
            return False

        if not employee.role.is_active:
            return False

        return employee.role.name in self.ALLOWED_ROLES

from rest_framework import status, viewsets
from rest_framework.response import Response

from .models import Employee

from common.filters import OrganizationFilterMixin
from common.constants import ROLE_LEVELS

class EmployeeViewSet(
    OrganizationFilterMixin,
    RolePermissionMixin,
    viewsets.ModelViewSet,
):
    
    permission_classes = [RolePermission]

    minimum_role_level = ROLE_LEVELS[RoleCodes.CENTER_MANAGER]

    required_permission = "employee.create"
    """
    API endpoints for Employee Management.
    """

    queryset = Employee.objects.all().order_by("employee_code")
    serializer_class = EmployeeSerializer
    

    def create(self, request, *args, **kwargs):
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        result = EmployeeService.create_employee(
            serializer.validated_data
        )

        employee = result["employee"]
        temporary_password = result["temporary_password"]

        response_serializer = self.get_serializer(employee)

        return Response(
            {
                "message": "Employee created successfully.",
                "employee": response_serializer.data,
                "login_credentials": {
                    "username": employee.user.username,
                    "temporary_password": temporary_password,
                },
            },
            status=status.HTTP_201_CREATED,
        )

    created = 0

    for role in DEFAULT_ROLES:
        _, is_created = Role.objects.get_or_create(
            code=role["code"],

    "NUTRITIONIST": [
        "member.view",
    ],

    "HOUSEKEEPING": [],

def assign_default_permissions():
    assigned_roles = 0

    for role_name, permission_codes in ROLE_PERMISSION_MAP.items():

        role = Role.objects.filter(code=role_name).first()

        if not role:
            print(f"⚠️ Role '{role_name}' not found. Skipping...")
            continue

        if permission_codes == "__ALL__":
            role.permissions.set(Permission.objects.all())
        else:
            permissions = Permission.objects.filter(
                code__in=permission_codes
            )
            role.permissions.set(permissions)

        assigned_roles += 1

    return assigned_roles

from django.db import transaction

from roles.seeds import seed_permissions
from bootstrap.roles import seed_roles
from bootstrap.role_permissions import assign_default_permissions


class BootstrapService:
    """
    Initializes FitForge master data.
    Safe to run multiple times.
    """

    @staticmethod
    @transaction.atomic
    def run():

        result = {}

        created_permissions = seed_permissions()
        created_roles = seed_roles()

        assigned_roles = assign_default_permissions()

        result["permissions_created"] = created_permissions
        result["roles_created"] = created_roles
        result["roles_configured"] = assigned_roles

        return result
