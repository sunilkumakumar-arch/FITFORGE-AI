from django.db import models


class Organization(models.Model):
    name = models.CharField(max_length=200)
    owner_name = models.CharField(max_length=200)
    email = models.EmailField(unique=True)
    phone = models.CharField(max_length=15)

    created_at = models.DateTimeField(
        auto_now_add=True,
    )

    updated_at = models.DateTimeField(
        auto_now=True,
    )

    is_active = models.BooleanField(
        default=True,
    )

    def __str__(self):
        return self.name


class Branch(models.Model):
    organization = models.ForeignKey(
        Organization,
        on_delete=models.CASCADE,
        related_name="branches"
    )

    name = models.CharField(max_length=200)
    city = models.CharField(max_length=100)
    address = models.TextField()

    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        verbose_name = "Branch"
        verbose_name_plural = "Branches"

    def __str__(self):
        return f"{self.organization.name} - {self.name}"



    

class Department(models.Model):

    name = models.CharField(max_length=100, unique=True)

    description = models.TextField(blank=True)

    is_active = models.BooleanField(default=True)

    created_at = models.DateTimeField(auto_now_add=True)

    updated_at = models.DateTimeField(auto_now=True)

    def __str__(self):
        return self.name    
    
class Designation(models.Model):

    department = models.ForeignKey(
        Department,
        on_delete=models.CASCADE,
        related_name="designations",
    )

    name = models.CharField(
        max_length=100,
    )

    description = models.TextField(
        blank=True,
    )

    hierarchy_level = models.PositiveIntegerField(
        default=1,
        help_text="Lower number indicates higher designation level.",
    )

    is_active = models.BooleanField(
        default=True,
    )

    created_at = models.DateTimeField(
        auto_now_add=True,
    )

    updated_at = models.DateTimeField(
        auto_now=True,
    )

    class Meta:
        ordering = ["department", "hierarchy_level", "name"]
        unique_together = ("department", "name")

    def __str__(self):
        return f"{self.department.name} - {self.name}"    


class Shift(models.Model):
    SHIFT_TYPE_CHOICES = [
        ("GENERAL", "General"),
        ("MORNING", "Morning"),
        ("EVENING", "Evening"),
        ("NIGHT", "Night"),
        ("FLEXIBLE", "Flexible"),
        ("SPLIT", "Split"),
    ]

    name = models.CharField(
        max_length=100,
        unique=True,
    )

    code = models.CharField(
        max_length=20,
        unique=True,
    )

    shift_type = models.CharField(
        max_length=20,
        choices=SHIFT_TYPE_CHOICES,
        default="GENERAL",
    )

    start_time = models.TimeField()

    end_time = models.TimeField()

    grace_period_minutes = models.PositiveIntegerField(
        default=15,
    )

    break_duration_minutes = models.PositiveIntegerField(
        default=30,
    )

    minimum_working_hours = models.DecimalField(
        max_digits=4,
        decimal_places=2,
        default=8.00,
    )

    weekly_working_days = models.PositiveSmallIntegerField(
        default=6,
    )

    is_active = models.BooleanField(
        default=True,
    )

    created_at = models.DateTimeField(
        auto_now_add=True,
    )

    updated_at = models.DateTimeField(
        auto_now=True,
    )

    class Meta:
        ordering = ["name"]

    def __str__(self):
        return f"{self.name} ({self.code})"

from rest_framework import serializers

from .models import Organization, Branch



class OrganizationSerializer(serializers.ModelSerializer):
    class Meta:
        model = Organization
        fields = [
            "id",
            "name",
            "owner_name",
            "email",
            "phone",
            "is_active",
            "created_at",
            "updated_at",
        ]

        read_only_fields = (
            "id",
            "created_at",
            "updated_at",
        )

    def validate_name(self, value):
        value = value.strip()

        if len(value) < 3:
            raise serializers.ValidationError(
                "Organization name must contain at least 3 characters."
            )

        return value

    def validate_owner_name(self, value):
        value = value.strip()

        if len(value) < 2:
            raise serializers.ValidationError(
                "Owner name must contain at least 2 characters."
            )

        return value

    def validate_phone(self, value):
        if not value.isdigit():
            raise serializers.ValidationError(
                "Phone number must contain only digits."
            )

        if len(value) != 10:
            raise serializers.ValidationError(
                "Phone number must be exactly 10 digits."
            )

        return value


class BranchSerializer(serializers.ModelSerializer):
    class Meta:
        model = Branch
        fields = [
            "id",
            "organization",
            "name",
            "city",
            "address",
            "created_at",
        ]

        read_only_fields = (
            "id",
            "created_at",
        )

    def validate_name(self, value):
        value = value.strip()

        if len(value) < 3:
            raise serializers.ValidationError(
                "Branch name must contain at least 3 characters."
            )

        return value

    def validate_city(self, value):
        value = value.strip()

        if len(value) < 2:
            raise serializers.ValidationError(
                "City must contain at least 2 characters."
            )

        return value.title()

    def validate_address(self, value):
        value = value.strip()

        if len(value) < 5:
            raise serializers.ValidationError(
                "Address must contain at least 5 characters."
            )

        return value

from django.db import transaction

from audit.services import AuditService

from .models import Organization, Branch


class OrganizationService:
    """
    Business logic for Organization Management.
    """

    @staticmethod
    @transaction.atomic
    def create_organization(validated_data):
        """
        Create a new organization.
        """

        organization = Organization.objects.create(
            **validated_data,
        )

        AuditService.log(
            user=None,
            action="CREATE",
            module="ORGANIZATION",
            object_id=str(organization.id),
            object_name=organization.name,
            description=(
                f"Organization '{organization.name}' created successfully."
            ),
        )

        return organization

    @staticmethod
    @transaction.atomic
    def deactivate_organization(organization, user=None):
        """
        Soft delete an organization by marking it inactive.
        """

        organization.is_active = False
        organization.save(update_fields=["is_active"])

        AuditService.log(
            user=user,
            action="DELETE",
            module="ORGANIZATION",
            object_id=str(organization.id),
            object_name=organization.name,
            description=(
                f"Organization '{organization.name}' deactivated successfully."
            ),
        )

        return organization

from rest_framework import status, viewsets
from rest_framework.response import Response

from .models import Organization
from .serializers import OrganizationSerializer
from .services import OrganizationService
from roles.permissions import RolePermission
from common.responses import (
    created_response,
    success_response,
)

from django_filters.rest_framework import DjangoFilterBackend
from rest_framework.filters import SearchFilter, OrderingFilter

class OrganizationViewSet(viewsets.ModelViewSet):
    """
    API endpoints for Organization Management.
    """
    permission_classes = [RolePermission]
    permission_map = {
        "list": "organization.view",
        "retrieve": "organization.view",
        "create": "organization.create",
        "update": "organization.update",
        "partial_update": "organization.update",
        "destroy": "organization.delete",
    }

    queryset = Organization.objects.filter(
        is_active=True
    ).order_by("name")
    serializer_class = OrganizationSerializer

    filter_backends = (
        DjangoFilterBackend,
        SearchFilter,
        OrderingFilter,
    )

    filterset_fields = (
        "is_active",
    )

    search_fields = (
        "name",
        "owner_name",
        "email",
        "phone",
    )

    ordering_fields = (
        "name",
        "created_at",
    )

    ordering = (
        "name",
    )

    def create(self, request, *args, **kwargs):
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        organization = OrganizationService.create_organization(
            serializer.validated_data
        )

        response_serializer = self.get_serializer(organization)

        
        return created_response(
            data=response_serializer.data,
            message="Organization created successfully.",
        )

    def destroy(self, request, *args, **kwargs):
        """
        Soft delete an organization.
        """
        organization = self.get_object()

        OrganizationService.deactivate_organization(
            organization=organization,
            user=request.user if request.user.is_authenticated else None,
        )

        return success_response(
            message="Organization deactivated successfully.",
        )
